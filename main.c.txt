 void delete_all(ST **ptr)
 {
         if(*ptr==0)
         {
                 printf("no records found");
                return ;
        }
         ST *del=*ptr;
         while(del)
         {
                 *ptr =del->next;
                free(del);
                 del=*ptr;
         }
 }
 void reverse_link(ST **ptr)
 {
        if(*ptr==0)
        {
                 printf("no records found\n");
                 return;
         }
        int i,c=count(*ptr);
         ST **p,*t=*ptr;
         if(c>1)
         {
                 p=malloc(sizeof(ST *)*c);
                for(i=0;i<c;i++)
                 {
                        p[i]=t;
                         t=t->next;
                 }
                 for(i=c-1;i>0;i--)
                 {
                         p[i]->next=p[i-1];
                 }
                 p[0]->next=0;
                 *ptr=p[c-1];
         }
 }
 int count(ST *ptr)
 {
         int c=0;
         while(ptr)
         {
                 c++;
                 ptr=ptr->next;
         }
         return c;
    }

                 ptr=ptr->next;
         }
         printf("data saved in file\n");
         fclose(fp);
 }
