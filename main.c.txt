470 void reverse_link(ST **ptr)
471 {
472         if(*ptr==0)
473         {
474                 printf("no records found\n");
475                 return;
476         }
477         int i,c=count(*ptr);
478         ST **p,*t=*ptr;
479         if(c>1)
480         {
481                 p=malloc(sizeof(ST *)*c);
482                 for(i=0;i<c;i++)
483                 {
484                         p[i]=t;
485                         t=t->next;
486                 }
487                 for(i=c-1;i>0;i--)
488                 {
489                         p[i]->next=p[i-1];
490                 }
491                 p[0]->next=0;
492                 *ptr=p[c-1];
493         }
494 }
